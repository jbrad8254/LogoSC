cmake_minimum_required(VERSION 3.20)
project(logosc_knot_grid VERSION 0.1.0 LANGUAGES CXX)
add_executable(logosc-knot-grid src/main.cpp)
target_compile_features(logosc-knot-grid PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(logosc-knot-grid PRIVATE /W4 /permissive- /EHsc)
else()
    target_compile_options(logosc-knot-grid PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    add_test(NAME logosc-knot-grid-self-test COMMAND logosc-knot-grid --self-test)
endif()
install(TARGETS logosc-knot-grid RUNTIME DESTINATION bin)
